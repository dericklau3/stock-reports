    window.isInCart = function (packageId) {
                    const cartCookieName = window.getCartCookieName();
            const guestCart = window.getCookie(cartCookieName);
            if (guestCart) {
                try {
                    const cartData = JSON.parse(guestCart);

        } else {
            const expires = new Date();

    function mergeCartData(existingItems, newItems) {
        const cartMap = new Map();
        existingItems.forEach(function (item) {
            if (item.package_id) cartMap.set(String(item.package_id), item);
        });
        newItems.forEach(function (item) {
            if (item.package_id && !cartMap.has(String(item.package_id))) {
                cartMap.set(String(item.package_id), item);
            }
        });
        return Array.from(cartMap.values());
    }

                if (!url || url.indexOf('/api/cart/toList') === -1) return url;

                if (!guest) return url;
                if (url.indexOf('guest_id=') !== -1) return url;
                var sep = url.indexOf('?') === -1 ? '?' : '&';
                return url + sep + 'guest_id=' + encodeURIComponent(guest);
            } catch (e) { return url; }

DRAM Industry Revenue Rises 59.5% QoQ in 2Q26 as Supply Expansion Continues to Lag Demand Growth, Says TrendForce
7 September 2026
Semiconductors
TrendForce
TrendForce’s latest findings on the memory industry reveal that sharp rises in traditional DRAM contract prices have led to a 59.5% QoQ increase in total DRAM industry revenue, reaching nearly US$154.73 billion in 2Q26.
Growing demand for AI servers, fueled by expanding LLM training and AI inference, has driven higher shipments of HBM3e, LPDDR5X, and high-capacity RDIMMS. The surge in agentic AI applications is further boosting RDIMM demand across various capacities.
On the supply side, suppliers’ inventories remain at historic lows, with additional supply mainly allocated for server use. Consequently, DRAM bit shipments grew only modestly in the second quarter.
Looking ahead, modest growth in bit shipments is expected to be one of the key factors affecting third-quarter revenue. Conventional DRAM contract price growth, however, is projected to moderate to 13–18% QoQ. This reflects a shift in some demand from high-capacity RDIMMs toward lower-capacity products, as well as the limited ability of PC and smartphone customers to absorb further price increases. Consumer DRAM is expected to post the strongest price growth, as suppliers have significantly curtailed supply.
TrendForce notes that Samsung posted the strongest QoQ bit shipment growth among the three major suppliers in 2Q26, supported by its early move into HBM4 mass production and shipments. Combined with a substantial increase in ASP, this lifted Samsung’s quarterly revenue to $60.98 billion, up 63.4% QoQ. The company ranked first with a market share of 39.4%.
SK hynix ranked second. Since HBM accounted for the highest proportion of its total bit shipments among the top three suppliers, growth in ASP was relatively limited. Its 2Q26 revenue increased 37.9% QoQ to $38.59 billion, while market share edged down to 24.9%.
Micron continued to prioritize higher-priced server DRAM in its product mix amid capacity constraints. Revenue rose 65.5% QoQ to $36.0 billion in 2Q26, lifting its market share slightly to 23.3%. TrendForce notes that from 2026 to 2027, the three major suppliers will primarily increase bit supply by accelerating migrations to advanced processes.
Meanwhile, wafer starts are expected to see only modest increases through production process optimization, cleanroom acquisitions, and the reallocation of cleanroom space from other product lines.
By comparison, suppliers such as Nanya, Winbond, and PSMC remain focused on mature-process products, with their 2Q26 growth continuing to benefit from demand that the three major suppliers can no longer fully serve as they migrate to more advanced processes. Significant increases in DDR4 and DDR3 contract prices drove Nanya’s revenue up 68.3% QoQ to $2.612 billion. Winbond’s revenue increased 75.8% QoQ to $998 million.
PSMC’s reported DRAM revenue, primarily derived from consumer DRAM products, surged 167.8% QoQ to $115 million in 2Q26. Including foundry revenue, its total revenue increased 54% QoQ. Looking ahead, the company is expected to begin its next phase of supply expansion after obtaining a process technology license from Micron.

                try {

                        // 檢查是否有 hostname 且與當前網站不同
                        if (link.hostname && link.hostname !== window.location.hostname) {
                            link.target = '_blank';
                            // 安全性最佳實踐：加上 noopener noreferrer
                            link.rel = 'noopener noreferrer';
                        }
                    });
                } catch (e) {
                    console.error('Error handling external links:', e);
                }

                
                if (window.matchMedia("(max-width: 768px)").matches) {
                    console.log('Mobile view detected');
                    
                    // 遍歷<article>內的<p>，尋找圖片
                    const articleParagraphs = document.querySelectorAll("article p");
                    console.log('Found paragraphs:', articleParagraphs.length);
                    
                    articleParagraphs.forEach(p => {
                        const img = p.querySelector("img");
                        if (!img) return; 

                        console.log('Found image in paragraph:', img.src);
                        
                        // 給放大鏡圖示CSS定位用 class
                        p.classList.add("has-img-zoom");
                        console.log('Added has-img-zoom class to paragraph');

                        // 防止重複綁定事件
                        if (p.dataset.zoomBound) return;
                        p.dataset.zoomBound = "true";

                        // 監聽圖片區塊的點擊事件
                        p.addEventListener("click", function (e) {
                            const rect = p.getBoundingClientRect();
                            const iconSize = 40; // 放大鏡圖示的觸控範圍
                            
                            // 計算滑鼠/手指點擊位置相對於圖片元素的座標
                            const x = e.clientX - rect.left;
                            const y = e.clientY - rect.top;
                            
                            // 判定是否點擊在右上角的放大鏡圖示範圍內
                            const clickedInIcon = (x >= rect.width - iconSize) && (y <= iconSize + 10);
                            
                            console.log('Click detected:', {x, y, clickedInIcon});
                            
                            if (!clickedInIcon) return;

                            e.preventDefault();
                            e.stopPropagation();
                            openLightbox(img.src, img.alt || ""); // 開啟燈箱
                        });
                    });

                    // 開啟燈箱
                    function openLightbox(src, alt) {
                        let lightbox = document.getElementById("img-lightbox");
                        const hostname = window.location.hostname;
                        const isTwSite = hostname.includes(".com.tw"); //中文站
                        let txtZoomIn, txtZoomOut;

                        if (isTwSite) {
                            // 台灣站 (.com.tw) -> 使用中文
                            txtZoomIn = "點圖放大";
                            txtZoomOut = "點圖縮小";
                        } else {
                            // 全球站 (.com) 或其他 -> 使用英文 (強調點擊圖片)
                            txtZoomIn = "Tap Image to Zoom";
                            txtZoomOut = "Tap Image to Reset";
                        }

                        // --- 2. 如果燈箱 DOM 不存在，則動態建立 (只執行一次) ---
                        if (!lightbox) {
                            lightbox = document.createElement("div");
                            lightbox.id = "img-lightbox";

                            // 建立頂部提示文字框
                            const hint = document.createElement("div");
                            hint.className = "hint-text";
                            hint.innerText = txtZoomIn; // 預設顯示放大提示
                            lightbox.appendChild(hint);

                            // 建立圖片容器 (負責 CSS Grid 置中與捲動功能)
                            const container = document.createElement("div");
                            container.className = "img-container"; 
                            
                            const enlargedImg = document.createElement("img");
                            container.appendChild(enlargedImg);
                            lightbox.appendChild(container);

                            // 建立右上角關閉按鈕 (X)
                            const closeBtn = document.createElement("div");
                            closeBtn.className = "close-btn";
                            closeBtn.innerHTML = "&times;";
                            
                            // 定義關閉行為函式
                            const closeAction = () => {
                                lightbox.style.display = "none";
                                document.body.style.overflow = ""; // 恢復背景網頁捲動
                                
                                // 重置所有狀態：移除放大 class、捲動軸歸零、清空圖片來源
                                container.classList.remove("zoomed");
                                container.scrollTop = 0;
                                container.scrollLeft = 0;
                                enlargedImg.src = ""; 
                                
                                // 文字重置回初始狀態
                                hint.innerText = txtZoomIn;
                            };

                            closeBtn.addEventListener("click", closeAction);
                            lightbox.appendChild(closeBtn);
                            document.body.appendChild(lightbox);
                            
                            // --- 3. 圖片點擊互動：切換放大/縮小 ---
                            // 使用單一事件監聽器處理 lightbox 內的所有點擊
                            lightbox.addEventListener("click", (e) => {
                                // 判斷點擊的是否為圖片
                                if (e.target === enlargedImg) {
                                    e.stopPropagation(); 
                                    
                                    // 圖片點擊邏輯：切換放大/縮小
                                    if (container.classList.contains("zoomed")) {
                                        // 當前是放大 -> 執行縮小
                                        container.classList.remove("zoomed");
                                        hint.innerText = txtZoomIn;
                                        
                                        // 強制重置樣式，確保圖片縮回適合螢幕的大小且不佔用額外空間
                                        enlargedImg.style.width = "auto";
                                        enlargedImg.style.height = "auto";
                                        enlargedImg.style.maxWidth = "100%";
                                        enlargedImg.style.maxHeight = "100%";
                                    } else {
                                        // 當前是縮小 -> 執行放大
                                        container.classList.add("zoomed");
                                        hint.innerText = txtZoomOut;
                                        
                                        // 清除 inline 樣式，讓 CSS 類別 (.zoomed img) 控制放大效果
                                        enlargedImg.style.width = "";
                                        enlargedImg.style.height = "";
                                        enlargedImg.style.maxWidth = "";
                                        enlargedImg.style.maxHeight = "";
                                    }
                                } else {
                                    // 點擊圖片以外的任何區域（Lightbox 背景、Container 空白處） -> 執行關閉
                                    closeAction();
                                }
                            });
                        }

                        // --- 4. 每次開啟燈箱時的初始化設定 ---
                        const hint = lightbox.querySelector(".hint-text");
                        if(hint) hint.innerText = txtZoomIn;

                        const enlargedImg = lightbox.querySelector("img");
                        const container = lightbox.querySelector(".img-container");
                        
                        // 設定圖片來源與替代文字
                        enlargedImg.src = src;

                        container.scrollTop = 0;
                        container.scrollLeft = 0;

                        // 顯示燈箱並鎖定背景滾動
                        lightbox.style.display = "block";
                        document.body.style.overflow = "hidden";
                    }
                } else {
                    console.log('Desktop view - script not running');
                }
            });
